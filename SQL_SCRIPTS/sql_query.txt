CREATE TABLE PRODUCT (
    P_CODE      VARCHAR(10)   NOT NULL,
    P_DESCRIPT  VARCHAR(35)   NOT NULL,
    P_INDATE    DATE          NOT NULL,
    P_QOH       SMALLINT      NOT NULL,
    P_MIN       SMALLINT      NOT NULL,
    P_PRICE     DECIMAL(8,2)  NOT NULL,
    P_DISCOUNT  DECIMAL(5,2)  NOT NULL,
    V_CODE      INTEGER,
    PRIMARY KEY (P_CODE),
    FOREIGN KEY (V_CODE) REFERENCES VENDOR (V_CODE)
);

CREATE TABLE LINE (
INV_NUMBER VARCHAR(10) NOT NULL,
LINE_NUMBER NUMERIC(2,0) NOT NULL,
P_CODE VARCHAR(10) NOT NULL,
LINE_UNITS NUMERIC(9,2) NOT NULL DEFAULT 0,
LINE_PRICE NUMERIC(9,2) NOT NULL DEFAULT 0,
PRIMARY KEY (INV_NUMBER, LINE_NUMBER),
FOREIGN KEY (P_CODE) REFERENCES PRODUCT (P_CODE)
);


CREATE TABLE CUSTOMER(
    CUS_CODE INT PRIMARY KEY,
    CUS_LNAME VARCHAR(15) NOT NULL,
    CUS_FNAME VARCHAR(15) NOT NULL, 
    CUS_INITIAL CHAR(1),
    CUS_AREACODE CHAR(3) DEFAULT '615' NOT NULL,
    CHECK(CUS_AREACODE IN ('615','713','931')),
    CUS_PHONE CHAR(8) NOT NULL,
    CUS_BALANCE DECIMAL(9,2) DEFAULT 0.00,
    CONSTRAINT CUS_UT1 UNIQUE (CUS_LNAME, CUS_FNAME)
);


CREATE TABLE INVOICE(
    INV_NUMBER INT PRIMARY KEY,
    CUS_CODE INT NOT NULL REFERENCES CUSTOMER(CUS_CODE),
    INV_DATE TIMESTAMP NOT NULL,
    CONSTRAINT INV_CK1 CHECK (INV_DATE > str_to_date ('01-JAN-2010','DD-MM-YYYY'))
);

CREATE TABLE LINE(
    INV_NUMBER INT NOT NULL,
    LINE_NUMBER DECIMAL(2,0) NOT NULL,
    P_CODE VARCHAR(10) NOT NULL,
    LINE_UNITS DECIMAL(9,2) DEFAULT 0.00 NOT NULL,
    LINE_PRICE DECIMAL(9,2) DEFAULT 0.00 NOT NULL,
    PRIMARY KEY (INV_NUMBER, LINE_NUMBER),
    FOREIGN KEY (P_CODE) REFERENCES PRODUCT(P_CODE),
    CONSTRAINT LINE_UI1 UNIQUE(INV_NUMBER, P_CODE)
);
